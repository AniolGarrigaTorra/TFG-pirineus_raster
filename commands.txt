1: Make grid
python -m src.make_grid \
  --project-config configs/project.yaml \
  --aoi-config configs/aoi/experimental_pallars_sobira.yaml \
  --resolution 100

1.1: Validate grid
python -m src.validation.validate_grid \
  --project-config configs/project.yaml \
  --aoi-config configs/aoi/experimental_pallars_sobira.yaml \
  --resolution 100

2.1: run config
pirineus-raster run configs/runs/pallars_worldclim_100m.yaml

2.2: debugging
pirineus-raster run-source \
  --source-config configs/sources/worldclim/worldclim_v2_1_climate_normals.yaml \
  --stage build

3: Inspect the dataset
pirineus-raster inspect data_processed/datasets/pallars_worldclim_100m/

4: Validate the dataset
pirineus-raster validate-dataset data_processed/datasets/pallars_worldclim_100m/

5: Config workbench CLI
pirineus-raster catalog --source-config configs/sources/worldclim/worldclim_cmip6_future.yaml
pirineus-raster validate-config configs/runs/pallars_worldclim_cmip6_simplified_100m.yaml
pirineus-raster render-run configs/runs/pallars_worldclim_cmip6_simplified_100m.yaml

6: React workbench
pirineus-raster serve-config-api --host 127.0.0.1 --port 8765
cd ui
npm install
npm run dev


HPC JOBS:

sbatch jobs/run_raster_features.sh configs/runs/pallars_worldclim_full_100m.yaml

squeue -u $USER

squeue -j <JOB_ID>

Control the job:
scontrol show job <JOB_ID>

See log:
tail -f logs/*2611545*.out

See errors:
cat logs/pirineus_raster_<JOB_ID>.err

See final state:
sacct -j <JOB_ID> --format=JobID,JobName,State,ExitCode,Elapsed,MaxRSS,AllocCPUS,NodeList
Cancel:
scancel <JOB_ID>
